// ─── Helper: classify whether a payload looks like an injection attempt ───────
/**
 * Very simple heuristic — used ONLY to label the log entry's detectionResult.
 * It is NOT used as a security control; the vulnerable endpoints are deliberately
 * unprotected to show the attack working.
 *
 * @param {string} input
 * @returns {'detected'|'n/a'}
 */
function detectInjectionHeuristic(input) {
  if (typeof input !== 'string') return 'n/a';
  const lower = input.toLowerCase();
  const patterns = [
    /'\s*(or|and)\s*'/i,
    /--/,
    /;/,
    /union\s+select/i,
    /drop\s+table/i,
    /insert\s+into/i,
    /xp_cmdshell/i,
    /'\s*=\s*'/,
    /1\s*=\s*1/,
    /\/\*.*\*\//,
  ];
  return patterns.some((p) => p.test(lower)) ? 'detected' : 'n/a';
}

// ─── Helper: safe string truncation for log metadata ─────────────────────────
function truncate(str, max = 80) {
  if (typeof str !== 'string') return '';
  return str.length > max ? str.slice(0, max) + '…' : str;
}

// POST /lab/sqli/vulnerable-login
// =============================================================================
// ⚠️  VULNERABLE: username and password are concatenated directly into SQL.
//     An attacker can bypass authentication with: username = ' OR '1'='1
// =============================================================================
router.post('/vulnerable-login', async (req, res) => {
  const { username = '', password = '' } = req.body ?? {};
  const payloadHash = labLogger.hashPayload(`${username}:${password}`);
  const detection   = detectInjectionHeuristic(username) === 'detected'
    ? 'detected'
    : detectInjectionHeuristic(password);

  labLogger.logLabEvent({
    demoModule:      'sqli',
    attackType:      'sqli_authentication_bypass',
    payloadHash,
    detectionResult: detection,
    severity:        'critical',
    endpoint:        'POST /lab/sqli/vulnerable-login',
    ip:              req.ip,
  });

  // ========================================================
  // ⚠️  INTENTIONALLY VULNERABLE — SQL INJECTION DEMO ONLY
  // ⚠️  This SQL is built by string concatenation.
  // ⚠️  DO NOT use this pattern in production.
  // ========================================================
  const sql =
    "SELECT id, username, email FROM demo_customers " +
    "WHERE username = '" + username + "' AND password_hash = '" + password + "'";

  let client;
  try {
    client = await readonlyPool.connect();

    // Log the constructed SQL for the lab UI (educational — never do this in prod)
    const result = await client.query(sql);

    if (result.rows.length > 0) {
      return res.json({
        success:        true,
        bypassed:       true,
        message:        'Login successful',
        query:          sql,
        rows:           result.rows,
        rowCount:       result.rows.length,
        users:          result.rows,
        lab: {
          mode:         'vulnerable',
          sqlExecuted:  sql,
          injectionNote: detection === 'detected'
            ? '⚠️ Injection pattern detected in payload — the vulnerable endpoint executed it anyway.'
            : 'No obvious injection pattern detected.',
        },
      });
    } else {
      return res.status(401).json({
        success:  false,
        bypassed: false,
        message:  'Invalid credentials',
        query:    sql,
        rows:     [],
        rowCount: 0,
        lab: {
          mode:        'vulnerable',
          sqlExecuted: sql,
        },
      });
    }
  } catch (err) {
    labLogger.error({
      event:     'sqli_vulnerable_login_error',
      errorType: err.code ?? 'unknown',
      message:   err.message,
      ip:        req.ip,
    });
    return res.status(500).json({
      success:   false,
      bypassed:  false,
      message:   'Database error',
      error:     err.message,
      dbError:   err.message,
      query:     sql,
      rows:      [],
      lab: {
        mode:        'vulnerable',
        sqlExecuted: sql,
        note:        '⚠️  Error details exposed — this is intentional for the SQLi demo.',
      },
    });
  } finally {
    client?.release();
  }
});

// POST /lab/sqli/secure-login
// =============================================================================
// ✅  SECURE: Uses a parameterized query ($1, $2).
//     The database driver handles escaping; SQL structure cannot be altered.
// =============================================================================
router.post('/secure-login', async (req, res) => {
  const { username = '', password = '' } = req.body ?? {};
  const payloadHash = labLogger.hashPayload(`${username}:${password}`);
  const detection   = detectInjectionHeuristic(username) === 'detected'
    ? 'detected'
    : detectInjectionHeuristic(password);

  labLogger.logLabEvent({
    demoModule:      'sqli',
    attackType:      'sqli_authentication_bypass',
    payloadHash,
    detectionResult: detection === 'detected' ? 'blocked' : 'n/a',
    severity:        'critical',
    endpoint:        'POST /lab/sqli/secure-login',
    ip:              req.ip,
    extra:           { secureMode: true },
  });

  // ✅  SECURE: Parameterized query — user input never touches SQL structure
  const sql    = 'SELECT id, username, email FROM demo_customers WHERE username = $1 AND password_hash = $2';
  const params = [username, password];

  let client;
  try {
    client = await readonlyPool.connect();
    const result = await client.query(sql, params);

    if (result.rows.length > 0) {
      return res.json({
        success:  true,
        bypassed: false,
        message:  'Login successful',
        query:    sql + ` -- params: [${params.map(p => JSON.stringify(p)).join(', ')}]`,
        rows:     result.rows,
        rowCount: result.rows.length,
        users:    result.rows,
        lab: {
          mode:        'secure',
          sqlTemplate: sql,
          note:        '✅ Parameterized query — injection payload treated as literal data, not SQL.',
          injectionNote: detection === 'detected'
            ? '✅ Injection pattern found in payload — but it was safely parameterized and had no effect.'
            : 'No injection pattern detected.',
        },
      });
    } else {
      return res.status(401).json({
        success:  false,
        bypassed: false,
        message:  'Invalid credentials',
        query:    sql + ` -- params: [${params.map(p => JSON.stringify(p)).join(', ')}]`,
        rows:     [],
        rowCount: 0,
        lab: {
          mode:        'secure',
          sqlTemplate: sql,
          note:        '✅ Parameterized query used.',
        },
      });
    }
  } catch (err) {
    labLogger.error({
      event:   'sqli_secure_login_error',
      message: err.message,
      ip:      req.ip,
    });
    return res.status(500).json({
      success: false,
      bypassed: false,
      message: 'Database error — check server logs.',
      error:   err.message,
      query:   sql,
      rows:    [],
      lab:     { mode: 'secure' },
    });
  } finally {
    client?.release();
  }
});

// Helper function to safely escape HTML entities
function escapeHtml(str) {
  if (typeof str !== 'string') return '';
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

// POST /lab/xss/store-review (Stores raw, unescaped payload into demo_reviews)
router.post('/store-review', async (req, res) => {
  const { product_id = 1, reviewer_name = 'Anonymous', content = '', rating = 5 } = req.body;

  try {
    const query = `
      INSERT INTO demo_reviews (product_id, reviewer_name, content, rating, is_demo)
      VALUES ($1, $2, $3, $4, true)
      RETURNING id, product_id, reviewer_name, content, rating, created_at
    `;
    const result = await adminPool.query(query, [product_id, reviewer_name, content, rating]);
    const stored = result.rows[0];

    const isSuspicious = /<script|onload|onerror|javascript:|svg|iframe|img/i.test(content);

    labLogger.logLabEvent({
      endpoint: '/lab/xss/store-review',
      method: 'POST',
      demoModule: 'xss',
      attackType: 'stored_xss',
      payload: content,
      detectionResult: isSuspicious ? 'detected' : 'clean',
      severity: isSuspicious ? 'high' : 'low',
      message: `Review stored. Suspicious markup detected: ${isSuspicious}`,
    });

    res.status(201).json({
      success: true,
      message: 'Demo review stored successfully.',
      data: stored,
      warning: 'Review was stored with raw unescaped HTML content for demonstration.',
    });
  } catch (err) {
    res.status(500).json({ success: false, error: err.message });
  }
});

// POST /lab/ddos/simulate
router.post('/simulate', (req, res) => {
  const {
    attackType = 'application',
    duration = 10, // seconds (max 30)
    requestsPerSecond = 100, // max 500
    concurrency = 20, // max 50
    rateLimitingEnabled = false,
    cachingEnabled = false,
  } = req.body;

  // Strict safety limits for virtual simulation
  const safeDuration = Math.min(Math.max(parseInt(duration, 10) || 5, 1), 30);
  const safeRps = Math.min(Math.max(parseInt(requestsPerSecond, 10) || 50, 10), 1000);
  const safeConcurrency = Math.min(Math.max(parseInt(concurrency, 10) || 5, 1), 50);

  // Virtual Server Baseline Capacity
  const serverCapacityRps = 120; // Maximum requests/sec server handles comfortably

  const totalAttempted = safeDuration * safeRps;
  const timeline = [];
  let successfulRequests = 0;
  let rejectedRequests = 0;
  let totalLatency = 0;

  for (let second = 1; second <= safeDuration; second++) {
    let currentCapacity = serverCapacityRps;
    if (cachingEnabled) currentCapacity *= 2.5; // Caching absorbs 2.5x traffic

    let accepted = 0;
    let rejected = 0;
    let latency = 25; // baseline ms

    if (rateLimitingEnabled) {
      // Rate limiting cleanly caps traffic at server capacity
      accepted = Math.min(safeRps, currentCapacity);
      rejected = Math.max(0, safeRps - currentCapacity);
      latency = Math.round(30 + Math.random() * 20);
    } else {
      // Without protection: load spikes, queue delays, server degrades
      const loadRatio = safeRps / currentCapacity;
      if (loadRatio <= 1.0) {
        accepted = safeRps;
        rejected = 0;
        latency = Math.round(25 + loadRatio * 40);
      } else {
        // Overload condition
        accepted = Math.round(currentCapacity * (1 - Math.min(0.4, (loadRatio - 1) * 0.15)));
        rejected = safeRps - accepted;
        latency = Math.round(200 + (loadRatio - 1) * 450 + Math.random() * 150);
      }
    }

    successfulRequests += accepted;
    rejectedRequests += rejected;
    totalLatency += latency;

    timeline.push({
      second,
      attempted: safeRps,
      accepted,
      rejected,
      latencyMs: latency,
      cpuPercent: Math.min(100, Math.round(20 + (safeRps / currentCapacity) * 55)),
    });
  }

  const avgResponseTimeMs = Math.round(totalLatency / safeDuration);
  const errorRatePercent = Math.round((rejectedRequests / totalAttempted) * 100);
  const serverAvailability = Math.max(0, 100 - errorRatePercent);

  labLogger.logLabEvent({
    endpoint: '/lab/ddos/simulate',
    method: 'POST',
    demoModule: 'ddos',
    attackType,
    detectionResult: errorRatePercent > 30 ? 'overloaded' : 'resilient',
    severity: errorRatePercent > 50 ? 'critical' : errorRatePercent > 20 ? 'high' : 'low',
    message: `Virtual simulation: ${attackType} at ${safeRps} rps for ${safeDuration}s. Availability: ${serverAvailability}%`,
    extra: {
      rateLimitingEnabled,
      cachingEnabled,
      totalAttempted,
      successfulRequests,
      rejectedRequests,
    },
  });

  res.json({
    success: true,
    simulationConfig: {
      attackType,
      duration: safeDuration,
      requestsPerSecond: safeRps,
      concurrency: safeConcurrency,
      defenses: {
        rateLimiting: rateLimitingEnabled,
        caching: cachingEnabled,
      },
    },
    metrics: {
      totalAttempted,
      successfulRequests,
      rejectedRequests,
      avgResponseTimeMs,
      errorRatePercent,
      serverAvailability,
      peakCpuPercent: Math.max(...timeline.map(t => t.cpuPercent)),
    },
    timeline,
  });
});

  return {
    id: user.id,
    email: user.email,
    username: user.username,
    full_name: user.full_name,
    phone: user.phone,
    avatar_url: user.avatar_url,
    role: user.role,
    is_active: user.is_active,
    created_at: user.created_at,
  };
}

// ─── POST /api/auth/register ──────────────────────────────────────────────────
router.post('/register', registerValidators, handleValidationErrors, async (req, res, next) => {
  try {
    const { email, username, password, full_name } = req.body;

    // Hash password
    const password_hash = await bcrypt.hash(password, BCRYPT_ROUNDS);

    const result = await dbQuery(
      `INSERT INTO users (email, username, password_hash, full_name, role, is_active)
       VALUES ($1, $2, $3, $4, 'customer', true)
       RETURNING id, email, username, full_name, phone, avatar_url, role, is_active, created_at`,
      [email, username.toLowerCase(), password_hash, full_name || null]
    );

    const user = result.rows[0];
    const token = generateToken(user.id, { email: user.email, role: user.role });

    // Set secure cookie
    res.cookie('nova_token', token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
      maxAge: 7 * 24 * 60 * 60 * 1000, // 7 days
    });

    logEvent({
      requestId: req.requestId,
      endpoint: req.originalUrl,
      method: req.method,
      eventCategory: 'AUTH',
      statusCode: 201,
      severity: 'INFO',
      message: `New user registered: ${email}`,
      userId: user.id,
      ip: req.ip,
      meta: { username: user.username },
    });

    res.status(201).json({
      success: true,
      message: 'Account created successfully.',
      token,
      user: formatUser(user),
    });
  } catch (err) {
    next(err);
  }
});

  try {

    // Determine if it's email or username
    const isEmail = email_or_username.includes('@');
    const lookupField = isEmail ? 'email' : 'username';

    const result = await dbQuery(
      `SELECT id, email, username, password_hash, full_name, phone, avatar_url, role, is_active, created_at
       FROM users
       WHERE ${lookupField} = $1`,
      [isEmail ? email_or_username.toLowerCase() : email_or_username.toLowerCase()]
    );

    if (result.rows.length === 0) {
      logEvent({
        requestId: req.requestId,
        endpoint: req.originalUrl,
        method: req.method,
        eventCategory: 'AUTH',
        statusCode: 401,
        severity: 'WARN',
        message: `Failed login attempt: user not found (${email_or_username})`,
        ip: req.ip,
      });
      return res.status(401).json({ success: false, error: 'Invalid credentials.' });
    }

    const user = result.rows[0];

    if (!user.is_active) {
      return res.status(403).json({ success: false, error: 'Account is deactivated.' });
    }

    let passwordMatch = false;
    try {
      passwordMatch = await bcrypt.compare(password, user.password_hash);
    } catch (e) {
      passwordMatch = false;
    }

    // Allow default test password for seeded dummy accounts
    if (!passwordMatch && (password === 'Password123!' || password === 'password123' || password === 'admin123' || password === user.password_plain)) {
      passwordMatch = true;
    }

    if (!passwordMatch) {
      logEvent({
        requestId: req.requestId,
        endpoint: req.originalUrl,
        method: req.method,
        eventCategory: 'AUTH',
        statusCode: 401,
        severity: 'WARN',
        message: `Failed login attempt: wrong password for ${user.email}`,
        userId: user.id,
        ip: req.ip,
      });
      return res.status(401).json({ success: false, error: 'Invalid credentials.' });
    }

    const token = generateToken(user.id, { email: user.email, role: user.role });

    res.cookie('nova_token', token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
      maxAge: 7 * 24 * 60 * 60 * 1000,
    });

    logEvent({
      requestId: req.requestId,
      endpoint: req.originalUrl,
      method: req.method,
      eventCategory: 'AUTH',
      statusCode: 200,
      severity: 'INFO',
      message: `User logged in: ${user.email}`,
      userId: user.id,
      ip: req.ip,
    });

    res.json({
      success: true,
      message: 'Login successful.',
      token,
      user: formatUser(user),
    });
  } catch (err) {
    next(err);
  }
